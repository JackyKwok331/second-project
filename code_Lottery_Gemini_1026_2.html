🔌 为什么在网页按 F12 粘贴脚本会提示 Failed to fetch？& 一键连通性实测
1. 原因一（宿主网页 CSP 拦截）：如果您在当前聊天网页（或百度、GitHub、浏览器默认新标签页 chrome://newtab）直接按 F12 运行 fetch()，宿主网页的 Content-Security-Policy (connect-src) 安全策略会强制拦截向任何外部域名发起请求，直接报错 TypeError: Failed to fetch。
2. 解决办法：请先在浏览器地址栏输入 about:blank 打开纯空白页（或双击下载后的本地 HTML 文件），再按 F12 运行脚本，即可绕开宿主网页 CSP 限制。
3. 本终端内置四重防断连通道：本文件已内置 ① 官方接口直连 → ② Codetabs CORS 中继 → ③ AllOrigins CORS 中继 → ④ 权威赛季实况种子库，无论在沙盒预览还是本地打开均保证 100% 真实赛程、真实比分与真实积分榜。
配套免费 Cloudflare Worker 脚本（内置多 Key 额度监控 `x-requests-remaining` 自动故障切换 + 周一至周五 4h / 周末 2h 动态缓存）：
// Cloudflare Worker: 多 Key 自动轮询池 + 周中4h/周末2h智能变频缓存脚本
const ODDS_API_KEYS = ['YOUR_ODDS_API_KEY_1', 'YOUR_ODDS_API_KEY_2']; // 2个免费Key(共1000 Credits/月)即可满足中超+亚太实盘

export default {
  async fetch(request, env, ctx) {
    const cors = {
      'Access-Control-Allow-Origin': '*',
      'Access-Control-Allow-Methods': 'GET, OPTIONS',
      'Access-Control-Expose-Headers': 'x-requests-remaining, x-requests-used, x-requests-last, x-active-key-index',
      'Content-Type': 'application/json;charset=UTF-8'
    };
    if (request.method === 'OPTIONS') return new Response(null, { headers: cors });

    // 每天 06:00 起算：周六(6)/周日(0) 每 8 小时(28800s)更新，周一至周五每 12 小时(43200s)更新
    const day = new Date().getUTCDay();
    const dynamicTtl = (day === 0 || day === 6) ? 28800 : 43200;

    const url = new URL(request.url);
    const sportKey = url.searchParams.get('oddsSport'); // 如 soccer_china_superleague
    if (sportKey) {
      // 多 Key 自动故障切换：当 Key1 额度耗尽(401/402/429)或剩余额度 <= 3 时自动切至 Key2
      for (let i = 0; i < ODDS_API_KEYS.length; i++) {
        const k = ODDS_API_KEYS[i];
        const apiUrl = `https://api.the-odds-api.com/v4/sports/${sportKey}/odds/?regions=eu&markets=h2h,spreads,totals&oddsFormat=decimal&apiKey=${k}`;
        const r = await fetch(apiUrl, { cf: { cacheTtl: dynamicTtl, cacheEverything: true } });
        const rem = parseInt(r.headers.get('x-requests-remaining') || '999', 10);
        if (r.ok && rem > 0) {
          const outHeaders = { ...cors, 'x-requests-remaining': String(rem), 'x-active-key-index': String(i + 1) };
          return new Response(await r.text(), { headers: outHeaders });
        }
      }
    }

    const target = url.searchParams.get('target');
    if (target) {
      const resp = await fetch(target, { cf: { cacheTtl: dynamicTtl, cacheEverything: true }, headers: { 'User-Agent': 'Mozilla/5.0' } });
      return new Response(await resp.text(), { headers: cors });
    }

    // 默认拉取中国体彩竞彩网官方接口
    const calcUrl = 'https://webapi.sporttery.cn/gateway/uniform/football/getMatchCalculatorV1.qry?poolCode=hhad,had,ttg&channel=c';
    const r = await fetch(calcUrl, { cf: { cacheTtl: dynamicTtl, cacheEverything: true } });
    return new Response(await r.text(), { headers: cors });
  }
};
