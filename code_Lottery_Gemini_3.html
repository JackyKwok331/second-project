export default {
  async fetch(request, env, ctx) {
    const corsHeaders = {
      'Access-Control-Allow-Origin': '*',
      'Access-Control-Allow-Methods': 'GET, OPTIONS',
      'Content-Type': 'application/json;charset=UTF-8'
    };
    if (request.method === 'OPTIONS') return new Response(null, { headers: corsHeaders });
    const url = new URL(request.url);
    const target = url.searchParams.get('target');
    if (target) {
      const resp = await fetch(target, { headers: { 'User-Agent': 'Mozilla/5.0', 'Accept': 'application/json' } });
      const body = await resp.text();
      return new Response(body, { headers: corsHeaders });
    }
    // 默认聚合拉取中国体彩竞彩网算奖器接口 + 赛果接口

    const r = await fetch(calcUrl, { cf: { cacheTtl: 7200, cacheEverything: true } });
    return new Response(await r.text(), { headers: corsHeaders });
  }
